# 写入 root crontab 的完整一行。cron 里 % 必须写成 \%，否则会被当成换行。
# master.sh / agent.sh / 安装包里的 install.sh 必须与这一行完全一致（忽略本说明和空行）。
# ntpdate 失败时向自建 auth.cdnfly.cn 要时间。不能用本机 date，那只是把错误时间再写回去。
*/10 * * * * /usr/sbin/ntpdate -u pool.ntp.org > /dev/null 2>&1 || (date_str=$(curl -fsSL --max-time 5 http://auth.cdnfly.cn/common/datetime) && printf \%s "$date_str" | grep -Eq "^[0-9]{4}-[0-9]{2}-[0-9]{2} [0-9]{2}:[0-9]{2}:[0-9]{2}\$" && timedatectl set-ntp false && timedatectl set-time "$date_str")
